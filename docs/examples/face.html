Face Plane & Emotion Examples

3D Perspective-n-Point face plane mapping, landmarks, and emotion evaluation

3D Mathematical Face Plane & Emotion HUD

Estimates head pose (pitch, yaw, roll), projects the face mathematical plane normal vector from the nose, and evaluates emotion probabilities.

Verified on videos/stock-face-headpose.mp4 and videos/face-emotion-01.mp4.

from yolozone import FaceTracker, PoseDetector
import cv2

pose = PoseDetector("yolov8s-pose.pt")
face_tracker = FaceTracker()

cap = cv2.VideoCapture("videos/face-emotion-01.mp4")

while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
        break

    keypoints = pose.find_keypoints(frame)
    tracked_faces = face_tracker.update(frame, keypoints=keypoints)

    # Render 3D mathematical plane axis, landmarks, and emotion HUD
    frame = face_tracker.draw_tracks(frame, tracked_faces, draw_plane=True, draw_trails=True)

    cv2.imshow("Face Plane & Emotion", frame)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
